import { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import styles from './PaymentModal.module.css';

const CURRENCIES = [
  { code: 'USD', symbol: '$', label: 'USD — US Dollar' },
  { code: 'GBP', symbol: '£', label: 'GBP — British Pound' },
  { code: 'EUR', symbol: '€', label: 'EUR — Euro' },
  { code: 'NGN', symbol: '₦', label: 'NGN — Nigerian Naira', multiplier: 1600 },
  { code: 'GHS', symbol: 'GH₵', label: 'GHS — Ghanaian Cedi', multiplier: 15 },
  { code: 'KES', symbol: 'KSh', label: 'KES — Kenyan Shilling', multiplier: 130 },
];

export default function PaymentModal({ reader, tier, onClose }) {
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [message, setMessage] = useState('');
  const [currency, setCurrency] = useState(CURRENCIES[0]);
  const [errors, setErrors] = useState({});
  const [isProcessing, setIsProcessing] = useState(false);

  const getAmount = () => {
    const m = currency.multiplier || 1;
    return Math.round(tier.priceNumber * m);
  };

  const validate = () => {
    const e = {};
    if (!name.trim()) e.name = 'Please enter your name';
    if (!email.trim() || !/\S+@\S+\.\S+/.test(email)) e.email = 'Please enter a valid email';
    return e;
  };

  const handleSubmit = async () => {
    const validationErrors = validate();
    if (Object.keys(validationErrors).length) {
      setErrors(validationErrors);
      return;
    }
    if (isProcessing) return;

    setIsProcessing(true);
    setErrors({});

    try {
      const response = await fetch('/.netlify/functions/initiate-payment', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          name,
          email,
          message,
          currency: currency.code,
          amount: getAmount(),
          reader: reader.name,
          tier: tier.name,
        }),
      });

      const data = await response.json();

      if (response.ok) {
        // Redirect user to the payment page
        window.location.href = data.link;
      } else {
        // Handle errors from the backend function
        setErrors({ api: data.error || 'An unexpected error occurred.' });
        setIsProcessing(false);
      }
    } catch (error) {
      // Handle network errors
      setErrors({ api: 'A network error occurred. Please try again.' });
      setIsProcessing(false);
    }
  };

  return (
    <AnimatePresence>
      <motion.div
        className={styles.backdrop}
        initial={{ opacity: 0 }}
        animate={{ opacity: 1 }}
        exit={{ opacity: 0 }}
        onClick={(e) => e.target === e.currentTarget && onClose()}
      >
        <motion.div
          className={styles.modal}
          initial={{ opacity: 0, y: 40, scale: 0.97 }}
          animate={{ opacity: 1, y: 0, scale: 1 }}
          exit={{ opacity: 0, y: 20, scale: 0.97 }}
          transition={{ duration: 0.35, ease: [0.22, 0.1, 0.36, 1] }}
        >
          <button className={styles.closeBtn} onClick={onClose} aria-label="Close">
            <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round">
              <line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/>
            </svg>
          </button>

          <div className={styles.inner}>
            <div className={styles.modalHeader}>
              <span className={styles.modalLabel}>Book a Reading</span>
              <h2 className={styles.modalTitle}>{tier.name} Session</h2>
              <p className={styles.modalSub}>
                with <strong>{reader.name}</strong> — <span className={styles.price}>{tier.price}</span>
              </p>
            </div>

            <div className={styles.form}>
              <div className={styles.field}>
                <label className={styles.label}>Your Name</label>
                <input
                  className={`${styles.input} ${errors.name ? styles.inputError : ''}`}
                  type="text"
                  placeholder="Full name"
                  value={name}
                  onChange={e => { setName(e.target.value); setErrors(p => ({ ...p, name: '' })); }}
                />
                {errors.name && <span className={styles.errMsg}>{errors.name}</span>}
              </div>

              <div className={styles.field}>
                <label className={styles.label}>Email Address</label>
                <input
                  className={`${styles.input} ${errors.email ? styles.inputError : ''}`}
                  type="email"
                  placeholder="your@email.com"
                  value={email}
                  onChange={e => { setEmail(e.target.value); setErrors(p => ({ ...p, email: '' })); }}
                />
                {errors.email && <span className={styles.errMsg}>{errors.email}</span>}
              </div>

              <div className={styles.field}>
                <label className={styles.label}>Currency</label>
                <select
                  className={styles.select}
                  value={currency.code}
                  onChange={e => setCurrency(CURRENCIES.find(c => c.code === e.target.value))}
                >
                  {CURRENCIES.map(c => (
                    <option key={c.code} value={c.code}>{c.label}</option>
                  ))}
                </select>
                <span className={styles.amountNote}>
                  You will pay: {currency.symbol}{getAmount().toLocaleString()} {currency.code}
                </span>
              </div>

              <div className={styles.field}>
                <label className={styles.label}>Message to {reader.name.split(' ')[0]} <span className={styles.optional}>(optional)</span></label>
                <textarea
                  className={styles.textarea}
                  placeholder="Share your main question or any context you want them to know..."
                  value={message}
                  onChange={e => setMessage(e.target.value.slice(0, 300))}
                  rows={3}
                />
                <span className={styles.charCount}>{message.length}/300</span>
              </div>
            </div>

            {errors.api && <div className={styles.apiError}>{errors.api}</div>}

            <button className={styles.proceedBtn} onClick={handleSubmit} disabled={isProcessing}>
              {isProcessing ? 'Processing...' : 'Proceed to Payment'}
              {!isProcessing && (
                <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                  <path d="M5 12h14M12 5l7 7-7 7"/>
                </svg>
              )}
            </button>
          </div>
        </motion.div>
      </motion.div>
    </AnimatePresence>
  );
}
